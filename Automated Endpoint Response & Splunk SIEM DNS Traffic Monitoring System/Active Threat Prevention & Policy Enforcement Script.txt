@echo off
:: دعم اللغة العربية والترميز العالمي
chcp 65001 >nul

:LOOP
:: البحث في عناوين النوافذ النشطة عن المواقع المحظورة وكلمات البحث
:: الكلمات: هكر، اختراق، يوتيوب، فيسبوك، نتفليكس
tasklist /V /FI "STATUS eq running" | findstr /I "هكر اختراق يوتيوب فيسبوك نتفليكس youtube facebook netflix" >nul

if %ERRORLEVEL% EQU 0 (
    :: إغلاق كافة المتصفحات (إيدج، كروم، فايرفوكس) فوراً
    taskkill /F /IM msedge.exe /T >nul 2>&1
    taskkill /F /IM chrome.exe /T >nul 2>&1
    taskkill /F /IM firefox.exe /T >nul 2>&1
    
    :: إظهار الرسالة التحذيرية للمستخدم
    msg * "تنبيه أمني: لقد حاولت الوصول إلى محتوى محظور أو البحث عن كلمات تخالف سياسة الشركة (هكر، اختراق، مواقع ترفيهية). تم إغلاق المتصفح فوراً."
)

:: فحص مستمر كل ثانية لضمان عدم إرهاق المعالج
timeout /t 1 /nobreak >nul
goto LOOP
